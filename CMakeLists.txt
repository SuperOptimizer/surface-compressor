cmake_minimum_required(VERSION 3.20)
project(surface_compressor VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(SFC_RELEASE_VERSION "${PROJECT_VERSION}-rc.1")
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(BUILD_SHARED_LIBS "Build a shared codec library" OFF)
option(SFC_TOOLS "Build TIFF conversion tools" ON)
option(SFC_TESTS "Build tests" ON)
option(SFC_FAST_MATH "Allow floating-point reassociation and contraction" ON)
option(SFC_SANITIZE "Enable address/undefined sanitizers" OFF)
add_library(surfcomp src/surfcomp.c)
add_library(surfcomp::surfcomp ALIAS surfcomp)
set_target_properties(surfcomp PROPERTIES POSITION_INDEPENDENT_CODE ON VERSION ${PROJECT_VERSION} SOVERSION 1)
target_include_directories(surfcomp PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(surfcomp PUBLIC c_std_11)
target_compile_definitions(surfcomp PRIVATE _FILE_OFFSET_BITS=64 _POSIX_C_SOURCE=200809L)
target_compile_options(surfcomp PRIVATE -Wall -Wextra -Werror )
if(SFC_FAST_MATH)
  target_compile_options(surfcomp PRIVATE -ffast-math -fno-finite-math-only)
endif()
target_link_libraries(surfcomp PUBLIC m)
if(SFC_SANITIZE)
  target_compile_options(surfcomp PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(surfcomp PUBLIC -fsanitize=address,undefined)
endif()
if(SFC_TOOLS)
  find_package(TIFF REQUIRED)
  add_executable(surface-compressor tools/main.c)
  target_link_libraries(surface-compressor PRIVATE surfcomp TIFF::TIFF)
endif()
if(SFC_TESTS)
  enable_testing()
  add_executable(test_surfcomp tests/test_codec.c)
  target_link_libraries(test_surfcomp PRIVATE surfcomp)
  add_test(NAME codec COMMAND test_surfcomp)
endif()
if(SFC_TESTS)
  add_executable(test_entropy tests/test_entropy.c)
  target_link_libraries(test_entropy PRIVATE surfcomp)
  if(SFC_FAST_MATH)

  endif()

endif()

option(SFC_NATIVE "Optimize for this build machine's CPU" OFF)
option(SFC_IPO "Enable supported interprocedural optimization in Release" OFF)
if(SFC_NATIVE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    target_compile_options(surfcomp PRIVATE -mcpu=native)
  else()
    target_compile_options(surfcomp PRIVATE -march=native)
  endif()
endif()
if(SFC_IPO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT sfc_ipo_supported OUTPUT sfc_ipo_error)
  if(NOT sfc_ipo_supported)
    message(FATAL_ERROR "SFC_IPO unavailable: ${sfc_ipo_error}")
  endif()
  set_property(TARGET surfcomp PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
if(SFC_TESTS)
  # Compile the opposite floating-point policy and exchange actual fixtures.
  add_library(surfcomp_cross STATIC src/surfcomp.c)
  target_link_libraries(surfcomp_cross PUBLIC surfcomp)
  if(NOT SFC_FAST_MATH)
    target_compile_options(surfcomp_cross PRIVATE -ffast-math -fno-finite-math-only)
  endif()
  add_executable(test_cross tests/test_codec.c)
  target_link_libraries(test_cross PRIVATE surfcomp_cross)
  add_test(NAME cross_write COMMAND test_surfcomp write ${CMAKE_CURRENT_BINARY_DIR}/cross-primary.bin)
  set_tests_properties(cross_write PROPERTIES FIXTURES_SETUP cross_primary)
  add_test(NAME cross_read COMMAND test_cross read ${CMAKE_CURRENT_BINARY_DIR}/cross-primary.bin)
  set_tests_properties(cross_read PROPERTIES FIXTURES_REQUIRED cross_primary)
  add_test(NAME cross_other_write COMMAND test_cross write ${CMAKE_CURRENT_BINARY_DIR}/cross-other.bin)
  set_tests_properties(cross_other_write PROPERTIES FIXTURES_SETUP cross_other)
  add_test(NAME cross_other_read COMMAND test_surfcomp read ${CMAKE_CURRENT_BINARY_DIR}/cross-other.bin)
  set_tests_properties(cross_other_read PROPERTIES FIXTURES_REQUIRED cross_other)

  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME fixture_manifest COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fixture_manifest.py)
    if(NOT SFC_SANITIZE)
      add_library(surfcomp_caller SHARED src/surfcomp.c)
      target_link_libraries(surfcomp_caller PRIVATE surfcomp)
      if(SFC_FAST_MATH)
        target_compile_options(surfcomp_caller PRIVATE -ffast-math -fno-finite-math-only)
      endif()
      add_test(NAME concurrent_caller COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_concurrent_caller.py $<TARGET_FILE:surfcomp_caller>)
      execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy"
        RESULT_VARIABLE sfc_numpy_available OUTPUT_QUIET ERROR_QUIET)
      if(sfc_numpy_available EQUAL 0)
        add_library(surfcomp_cross_caller SHARED src/surfcomp.c)
        target_link_libraries(surfcomp_cross_caller PRIVATE surfcomp)
        if(NOT SFC_FAST_MATH)
          target_compile_options(surfcomp_cross_caller PRIVATE -ffast-math -fno-finite-math-only)
        endif()
        add_test(NAME error_stress COMMAND ${Python3_EXECUTABLE}
          ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_error_stress.py
          $<TARGET_FILE:surfcomp_caller> $<TARGET_FILE:surfcomp_cross_caller>)
      endif()
    endif()
    if(SFC_TOOLS)
      execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy, tifffile, imagecodecs"
        RESULT_VARIABLE sfc_python_deps OUTPUT_QUIET ERROR_QUIET)
      if(sfc_python_deps EQUAL 0)
        add_test(NAME tiff_cli COMMAND ${Python3_EXECUTABLE}
          ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:surface-compressor>)
      else()
        message(STATUS "TIFF CLI tests not registered: Python numpy/tifffile/imagecodecs unavailable")
      endif()
    endif()
  endif()
endif()
if(SFC_TESTS)

  ${CMAKE_CURRENT_BINARY_DIR}/surfcompConfig.cmake

  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/surfcomp)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/surfcompConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/surfcompConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/surfcomp)

install(FILES spec/format.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/surfcomp/spec)

                   test_entropy test_transform test_failures fuzz_decode)

    target_compile_options(${sfc_target} PRIVATE -fno-finite-math-only)
  endif()
endforeach()
