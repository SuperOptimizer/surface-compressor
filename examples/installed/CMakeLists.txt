cmake_minimum_required(VERSION 3.20)
project(surfcomp_consumer LANGUAGES C CXX)
find_package(surfcomp 1 CONFIG REQUIRED)
add_executable(roundtrip roundtrip.c)
target_link_libraries(roundtrip PRIVATE surfcomp::surfcomp)
add_executable(cxx_header cxx_header.cpp)
target_link_libraries(cxx_header PRIVATE surfcomp::surfcomp)
enable_testing()
add_test(NAME roundtrip COMMAND roundtrip)
add_test(NAME cxx_header COMMAND cxx_header)
